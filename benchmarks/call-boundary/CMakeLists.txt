cmake_minimum_required(VERSION 3.20)
project(psizam_call_boundary LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(PSIZAM_ENABLE_LLVM ON CACHE BOOL "Build LLVM backend")
set(PSIZAM_ENABLE_TOOLS OFF CACHE BOOL "Build engine tools")
set(PSIZAM_ENABLE_TESTS OFF CACHE BOOL "Build full engine tests")
if(NOT PSIZAM_SOURCE_CHECKOUT)
  get_filename_component(PSIZAM_SOURCE_CHECKOUT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
endif()
add_subdirectory("${PSIZAM_SOURCE_CHECKOUT}" engine)
find_program(WAT2WASM wat2wasm REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/calls.wasm"
  COMMAND "${WAT2WASM}" "${CMAKE_CURRENT_SOURCE_DIR}/calls.wat" -o "${CMAKE_CURRENT_BINARY_DIR}/calls.wasm"
  DEPENDS calls.wat VERBATIM)
add_custom_target(call-fixture DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/calls.wasm")
add_executable(call-boundary calls.cpp)
add_dependencies(call-boundary call-fixture)
target_link_libraries(call-boundary PRIVATE psizam-exec)
if(PSIZAM_ENABLE_LLVM)
  target_link_libraries(call-boundary PRIVATE psizam-llvm)
endif()
if(WASMTIME_C_API)
  target_compile_definitions(call-boundary PRIVATE HAVE_WASMTIME)
  target_include_directories(call-boundary PRIVATE "${WASMTIME_C_API}/include")
  target_link_libraries(call-boundary PRIVATE "${WASMTIME_C_API}/lib/libwasmtime.a")
endif()
if(APPLE)
  execute_process(COMMAND brew --prefix llvm OUTPUT_VARIABLE llvm_prefix OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  target_link_options(call-boundary PRIVATE "-L${llvm_prefix}/lib/c++" "-Wl,-rpath,${llvm_prefix}/lib/c++")
endif()
enable_testing()
set(call_modes jit1 jit2)
if(PSIZAM_ENABLE_LLVM)
  list(APPEND call_modes llvm)
endif()
foreach(mode IN LISTS call_modes)
  foreach(policy none gas)
    add_test(NAME calls.${mode}.${policy} COMMAND call-boundary ${mode} "${CMAKE_CURRENT_BINARY_DIR}/calls.wasm" ${policy} check)
  endforeach()
endforeach()

add_executable(call-runtime runtime_bind.cpp "${PSIZAM_SOURCE_CHECKOUT}/src/runtime.cpp")
target_include_directories(call-runtime PRIVATE "${PSIZAM_SOURCE_CHECKOUT}/examples/composition")
target_link_libraries(call-runtime PRIVATE psizam-exec)
if(PSIZAM_ENABLE_LLVM)
  target_link_libraries(call-runtime PRIVATE psizam-llvm)
endif()
if(APPLE)
  target_link_options(call-runtime PRIVATE "-L${llvm_prefix}/lib/c++" "-Wl,-rpath,${llvm_prefix}/lib/c++")
endif()

set(call_test_target call-boundary)
set(call_test_wasm "${CMAKE_CURRENT_BINARY_DIR}/calls.wasm")
include(aot-tests.cmake)
