cmake_minimum_required(VERSION 3.20)
project(psizam_installed_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
option(TEST_LLVM "Exercise the optional installed LLVM factory" OFF)
if(TEST_LLVM)
   find_package(psizam CONFIG REQUIRED COMPONENTS LLVM)
else()
   find_package(psizam CONFIG REQUIRED)
endif()
add_executable(runtime-consumer main.cpp)
target_link_libraries(runtime-consumer PRIVATE psizam::psizam-exec)
if(TEST_LLVM)
   target_link_libraries(runtime-consumer PRIVATE psizam::psizam-llvm)
endif()
# Homebrew Clang's C++23 headers require its matching libc++ on macOS.
get_filename_component(_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
if(APPLE AND EXISTS "${_compiler_bin}/../lib/c++/libc++.dylib")
   target_link_options(runtime-consumer PRIVATE
      "-L${_compiler_bin}/../lib/c++" "-Wl,-rpath,${_compiler_bin}/../lib/c++")
endif()
