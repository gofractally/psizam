cmake_minimum_required(VERSION 3.20)
project(psizam_consumer LANGUAGES C CXX)
option(PSIZAM_CONSUMER_LLVM "Exercise the installed LLVM component" OFF)
if(PSIZAM_SOURCE_DIR)
  set(PSIZAM_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
  set(PSIZAM_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(PSIZAM_ENABLE_LLVM ${PSIZAM_CONSUMER_LLVM} CACHE BOOL "" FORCE)
  add_subdirectory(${PSIZAM_SOURCE_DIR} psizam)
else()
  if(PSIZAM_CONSUMER_LLVM)
    find_package(psizam COMPONENTS LLVM CONFIG REQUIRED)
  else()
    find_package(psizam CONFIG REQUIRED)
  endif()
endif()
add_executable(consumer main.cpp)
if(TARGET psizam-exec)
  target_link_libraries(consumer PRIVATE psizam-exec)
else()
  target_link_libraries(consumer PRIVATE psizam::psizam-exec)
endif()
enable_testing()
add_test(NAME consumer COMMAND consumer)

if(PSIZAM_CONSUMER_LLVM)
  target_compile_definitions(consumer PRIVATE PSIZAM_CONSUMER_LLVM)
  if(TARGET psizam-llvm)
    target_link_libraries(consumer PRIVATE psizam-llvm)
  else()
    target_link_libraries(consumer PRIVATE psizam::psizam-llvm)
  endif()
endif()
