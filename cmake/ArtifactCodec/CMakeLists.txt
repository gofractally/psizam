cmake_minimum_required(VERSION 3.20)
project(psizam_artifact_codec LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(psizam_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
if(TARGET psio::psio)
    # A containing isolated dependency project acquired the canonical pin.
elseif(FETCHCONTENT_SOURCE_DIR_PSIO)
    add_subdirectory("${FETCHCONTENT_SOURCE_DIR_PSIO}" psio)
else()
    find_package(psio 0.1 CONFIG REQUIRED)
endif()
if(FETCHCONTENT_SOURCE_DIR_UCC)
    add_library(ucc_headers INTERFACE)
    target_include_directories(ucc_headers INTERFACE "${FETCHCONTENT_SOURCE_DIR_UCC}/include")
else()
    find_package(ucc CONFIG REQUIRED)
    add_library(ucc_headers INTERFACE)
    target_link_libraries(ucc_headers INTERFACE ucc::ucc)
endif()
add_library(psizam-artifact-codec STATIC "${psizam_root}/src/native_artifact.cpp")
target_include_directories(psizam-artifact-codec PRIVATE "${psizam_root}/include"
    "${psizam_root}/external/softfloat/source/include")
# A private namespace is essential while old/new PSIO coexist in one process.
# Never propagate this macro, PSIO include paths or reflected PZAM types.
target_compile_definitions(psizam-artifact-codec PRIVATE
    psio=psizam_artifact_psio PSIZAM_SOFTFLOAT PSIZAM_DISABLE_EH)
target_compile_options(psizam-artifact-codec PRIVATE
    $<$<COMPILE_LANG_AND_ID:CXX,Clang,AppleClang>:-fbracket-depth=1024>)
target_link_libraries(psizam-artifact-codec PRIVATE psio::psio ucc_headers Threads::Threads)
set_target_properties(psizam-artifact-codec PROPERTIES
    UNITY_BUILD OFF DISABLE_PRECOMPILE_HEADERS ON)
install(TARGETS psizam-artifact-codec ARCHIVE DESTINATION lib)

# Fingerprint source content rather than only Git HEAD so explicit development
# overrides cannot reuse executable artifacts after an uncommitted repair.
set(native_identity "psizam-native-artifact-v1\n${PSIZAM_NATIVE_TOOLCHAIN_ID}\n${CMAKE_CXX_COMPILER_ID}\n${CMAKE_CXX_COMPILER_VERSION}\n${CMAKE_CXX_FLAGS}\n${CMAKE_CXX_FLAGS_RELEASE}\n${CMAKE_INTERPROCEDURAL_OPTIMIZATION}\n${CMAKE_SYSTEM_NAME}\n${CMAKE_SYSTEM_PROCESSOR}\n${CMAKE_SIZEOF_VOID_P}\n")
function(psizam_fingerprint_tree label directory)
    file(GLOB_RECURSE identity_files CONFIGURE_DEPENDS LIST_DIRECTORIES false "${directory}/*")
    list(SORT identity_files)
    set(value "${native_identity}")
    foreach(file IN LISTS identity_files)
        file(RELATIVE_PATH relative "${directory}" "${file}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
        file(SHA256 "${file}" digest)
        string(APPEND value "${label}/${relative}:${digest}\n")
    endforeach()
    set(native_identity "${value}" PARENT_SCOPE)
endfunction()
psizam_fingerprint_tree(runtime-headers "${psizam_root}/include")
psizam_fingerprint_tree(runtime-sources "${psizam_root}/src")
psizam_fingerprint_tree(softfloat "${psizam_root}/external/softfloat/source")
get_target_property(psio_identity_includes psio::psio INTERFACE_INCLUDE_DIRECTORIES)
set(identity_psio_found false)
foreach(include IN LISTS psio_identity_includes)
    string(REGEX REPLACE "^\\$<BUILD_INTERFACE:(.*)>$" "\\1" include "${include}")
    if(IS_ABSOLUTE "${include}" AND IS_DIRECTORY "${include}")
        psizam_fingerprint_tree(psio "${include}")
        set(identity_psio_found true)
    endif()
endforeach()
if(NOT identity_psio_found)
    message(FATAL_ERROR "Cannot fingerprint canonical PSIO headers for native artifacts")
endif()
if(FETCHCONTENT_SOURCE_DIR_UCC)
    psizam_fingerprint_tree(ucc "${FETCHCONTENT_SOURCE_DIR_UCC}/include")
else()
    get_target_property(ucc_identity_includes ucc::ucc INTERFACE_INCLUDE_DIRECTORIES)
    foreach(include IN LISTS ucc_identity_includes)
        if(IS_ABSOLUTE "${include}" AND IS_DIRECTORY "${include}")
            psizam_fingerprint_tree(ucc "${include}")
        endif()
    endforeach()
endif()
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" codec_build_digest)
string(APPEND native_identity "codec-cmake:${codec_build_digest}\n")
string(SHA256 native_build_identity "${native_identity}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/psizam-native-build.hpp"
    "#pragma once\n#define PSIZAM_NATIVE_BUILD_ID \"${native_build_identity}\"\n")
target_include_directories(psizam-artifact-codec PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
